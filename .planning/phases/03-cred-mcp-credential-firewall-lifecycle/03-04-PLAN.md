export function redactForDisplay<T>(obj: T): T;
// Deep-clones, then replaces every string value whose parent KEY matches
// isSecretFieldName(key) with "[REDACTED]".
// isSecretFieldName covers: "authorization", "cookie", "set-cookie",
//   "x-api-key", "x-auth-token", "api-key", plus the SECRET_FIELD_PATTERN regex.
// walkRedact descends into headers: { Authorization: "..." }
//   → isSecretFieldName("Authorization") → true → masked.
