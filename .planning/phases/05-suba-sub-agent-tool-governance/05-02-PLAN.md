    ```typescript
    /** Regex matching the SDK's exact "Tool not found" error format (agent-loop.js:356). */
    const NOT_FOUND_RE = /^Tool (\S+) not found$/;

    /**
     * Classify a tool that was not found during execution.
     * Returns enriched hint text for the error log.
     *
     * - "denylist": tool is in SUB_AGENT_TOOL_DENYLIST — never delegatable.
     * - "outside_profile": tool exists but is outside the active profile ceiling.
     */
    function classifyUnreachableTool(
      toolName: string,
      activeGroups: string[],
    ): string {
      if (SUB_AGENT_TOOL_DENYLIST.has(toolName)) {
        return `Tool '${toolName}' is denied to ALL sub-agents — the parent must perform this step.`;
      }
      // Find broader profiles that contain this tool
      const broadProfiles = Object.entries(TOOL_PROFILES)
        .filter(([name, tools]) => name !== "full" && !activeGroups.includes(name) && tools.includes(toolName))
        .map(([name]) => name);
      const suggestion = broadProfiles.length > 0
        ? broadProfiles.join("' | '")
        : "supervisor' or 'full";
      return (
        `Tool '${toolName}' is outside this sub-agent's profile. ` +
        `Re-spawn with tool_groups:['${suggestion}'].`
      );
    }
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-02-01 | Information Disclosure | classifyUnreachableTool logs tool names | accept | Tool names are internal identifiers, not secrets; no PII or credentials in tool names |
| T-05-02-02 | Tampering | NOT_FOUND_RE pattern too broad | mitigate | Regex `^Tool (\S+) not found$` is anchored (^ and $) and matches only the exact SDK format; freeform error text cannot spoof it |
| T-05-02-03 | Elevation of Privilege | classifyUnreachableTool used as oracle to enumerate denylist | accept | Denylist is not secret — it's published in @comis/core and in docs; the enrichment only triggers on existing tool names already visible to the running agent |
| T-05-02-04 | Spoofing | activeToolGroups passed with forged profile names | mitigate | activeToolGroups is set at daemon wiring time (PiExecutor setup), not from LLM or user input; profile names are validated against TOOL_PROFILES (only known profiles produce re-spawn hints) |
| T-05-02-05 | Denial of Service | enrichment on every tool_execution_end | accept | NOT_FOUND_RE.exec() is O(1) on a bounded string; negligible overhead; guard on `deps.activeToolGroups` short-circuits when not a sub-agent |
