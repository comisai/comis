    ```typescript
    // SPDX-License-Identifier: Apache-2.0
    /**
     * Port-backed MCP token store adapter (R8).
     * Wraps the existing createTokenStore (chokidar watch + 0600 preserved).
     * Syncs the OAuth token triple to OAuthCredentialStorePort on writes.
     * MUST live in daemon wiring — never in skills (skills→memory is forbidden).
     * @module
     */
    import type { OAuthCredentialStorePort, OAuthProfile } from "@comis/core";
    import { createTokenStore } from "@comis/skills/.../mcp-client/oauth/token-store.js";
    // NOTE: use the exact import path from the live codebase (read token-store.ts path)
    import type { TokenStore, TokenStoreDeps, OAuthTokens } from "@comis/skills/.../mcp-client/oauth/token-store.js";

    const MCP_OAUTH_PROVIDER = "mcp-oauth" as const;

    function mcpProfileId(serverName: string): string {
      return `${MCP_OAUTH_PROVIDER}:${serverName}`;
    }

    /**
     * Wraps createTokenStore with OAuthCredentialStorePort sync.
     * All TokenStore methods (including startWatch/stopWatch) delegate to the inner store.
     * The port receives the token triple (access/refresh/expires) on every saveTokens call.
     * Port write failure is NON-FATAL — the disk store is authoritative.
     */
    export function createPortBackedMcpTokenStore(
      port: OAuthCredentialStorePort,
      deps: TokenStoreDeps,
    ): TokenStore {
      const inner = createTokenStore(deps);  // chokidar watch + 0600 preserved
      return {
        ...inner,
        async saveTokens(serverName: string, tokens: OAuthTokens): Promise<void> {
          await inner.saveTokens(serverName, tokens);
          // Sync to port — port failure is non-fatal (disk store is authoritative source)
          const profileId = mcpProfileId(serverName);
          const profile: OAuthProfile = {
            provider: MCP_OAUTH_PROVIDER,
            profileId,
            access: tokens.access_token,
            refresh: tokens.refresh_token ?? "",
            expires: tokens.expires_at ?? 0,
            version: 1,
          };
          await port.set(profileId, profile).catch(() => {
            // Non-fatal: log at debug if logger available, then continue
          });
        },
        // startWatch and stopWatch are inherited from ...inner spread — they delegate to the
        // chokidar watch in the underlying createTokenStore. DO NOT override them.
      };
    }
    ```
